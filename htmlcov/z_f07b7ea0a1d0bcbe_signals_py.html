118 def handle_save(self, sender, instance, **kwargs): 

119 """ 

120 Handle save with a Celery task. 

121 

122 Given an individual model instance, update the object in the index. 

123 Update the related objects as well. 

124 

125 ``post_save`` fires before an enclosing transaction commits. Enqueuing the task immediately lets a 

126 worker pick it up and read the row on another connection before it is visible there, so the index 

127 update silently applies stale (or missing) data. Inside an atomic block, defer enqueuing to 

128 ``transaction.on_commit`` so the task never runs ahead of the transaction it depends on. 

129 """ 

130 if self.is_instance_indexed(instance): 

131 dispatch = partial(self.save.delay, *self.serialize_instance(instance)) 

132 if connection.in_atomic_block: 

133 transaction.on_commit(dispatch) 

134 else: 

135 dispatch()

142 doesn't exist anymore, and we can't get the related models instance. 

143 """

173 def is_instance_indexed(self, instance): 

174 """ 

175 Check whether a change to this instance can affect any document. 

176 

177 That is true both for models that have a document of their own and 

178 for models named in another document's ``related_models`` -- the 

179 latter have no document, yet a change to them must reindex the 

180 documents that embed them. Testing ``registry._models`` alone missed 

181 exactly that case, so the tasks that propagate related-model changes 

182 were never enqueued and embedded data went stale. ``registry`` 

183 answers the broader question via ``__contains__``. 

184 """ 

185 return instance.__class__ in registry

204 @staticmethod 

205 @shared_task() 

206 def delete(app_label, model_name, pk): 

207 """ 

208 Delete the document from the registry as a Celery task. 

209 

210 By the time a worker picks up a delete task, the row is already gone, so unlike ``save``/ 

211 ``delete_related`` this deliberately does not deserialize the instance from the database. 

212 ``registry.delete`` only needs the document id, and ``Document.generate_id`` derives that from the 

213 instance's pk alone, so an unsaved instance carrying just the pk is enough. 

214 

215 Celery's at-least-once delivery means the same delete can be dispatched twice, and the document may 

216 already be gone by the second attempt -- that specific 404 is expected and swallowed. Any other 

217 bulk error is re-raised, same as an unhandled error from ``save``/``delete_related`` would be, so 

218 it surfaces through Celery's own task failure/retry handling instead of disappearing into a log 

219 line that may not be routed anywhere. 

220 """

222 model = apps.get_model(app_label, model_name)

224 return
